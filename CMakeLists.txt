cmake_minimum_required(VERSION 3.20)
project(FeatherApi LANGUAGES CXX)
include(CTest)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(FeatherApi WIN32 main.cpp main_window.cpp storage.cpp http_client.cpp resources.rc app.manifest)
target_compile_definitions(FeatherApi PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(FeatherApi PRIVATE /utf-8 /W4 /WX)
endif()
target_link_libraries(FeatherApi PRIVATE comctl32 winhttp shlwapi shell32 ole32)

add_executable(FeatherApiSelfTests self_tests.cpp storage.cpp http_client.cpp)
target_compile_definitions(FeatherApiSelfTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(FeatherApiSelfTests PRIVATE /utf-8 /W4 /WX)
endif()
target_link_libraries(FeatherApiSelfTests PRIVATE winhttp shell32 ole32 ws2_32)
add_test(NAME FeatherApiSelfTests COMMAND FeatherApiSelfTests)

add_executable(FeatherApiUiTests ui_self_tests.cpp storage.cpp http_client.cpp resources.rc app.manifest)
target_compile_definitions(FeatherApiUiTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(FeatherApiUiTests PRIVATE /utf-8 /W4 /WX)
endif()
target_link_libraries(FeatherApiUiTests PRIVATE comctl32 comdlg32 winhttp shlwapi shell32 ole32)
add_test(NAME FeatherApiUiTests COMMAND FeatherApiUiTests --test)
